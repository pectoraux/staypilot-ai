   334→export const REVENUE_FORECAST_SERIES: Array<{ day: number; date: string; actual: number | null; predicted: number; lower: number; upper: number }> = (() => {

   337→  for (let i = -7; i <= 23; i++) {
   338→    const d = new Date(today); d.setDate(d.getDate() + i)
   339→    const weekend = d.getDay() === 5 || d.getDay() === 6 ? 1.35 : 1
   340→    const noise = 1 + (Math.sin(i / 2) * 0.08)
   341→    const predicted = Math.round(base * weekend * noise)
   342→    const lower = Math.round(predicted * 0.86)
   343→    const upper = Math.round(predicted * 1.16)
   344→    arr.push({ day: i, date: iso(d), actual: i < 0 ? Math.round(predicted * (0.92 + Math.random() * 0.16)) : null, predicted, lower, upper })
   345→  }
   346→  return arr
