   181→  const [sending, setSending] = React.useState(false)
   182→  const scrollRef = React.useRef<HTMLDivElement>(null)
   183→
   184→  const messages = agent ? histories[agent.id] ?? [] : []
   185→
   186→  React.useEffect(() => {
   187→    const el = scrollRef.current
   188→    if (el) el.scrollTop = el.scrollHeight
   189→  }, [messages.length, sending, agent?.id])
   190→
   191→  async function send(text: string) {
   192→    const trimmed = text.trim()
   193→    if (!trimmed || sending || !agent) return
   194→    const userMsg: AgentMsg = {
   195→      id: `u-${Date.now()}`,
   196→      role: 'user',
   197→      content: trimmed,
   198→      ts: now(),
   199→    }
   200→    const prior = histories[agent.id] ?? []
   201→    const next = [...prior, userMsg]
   202→    setHistories((h) => ({ ...h, [agent.id]: next }))
   203→    setInput('')
   204→    setSending(true)
   205→    try {
   206→      const history = next.map((m) => ({ role: m.role, content: m.content }))
   207→      const res = await fetch('/api/ai', {
   208→        method: 'POST',
   209→        headers: { 'Content-Type': 'application/json' },
   210→        body: JSON.stringify({
   211→          mode: 'agent-chat',
   212→          agentRole: `${agent.role} at ${PROPERTY.name}`,
   213→          message: trimmed,
   214→          history,
   215→        }),
   216→      })
   217→      const data = await res.json()
   218→      const reply =
   219→        data?.reply ??
   220→        `I'm ${agent.name}, your ${agent.role}. I'm reviewing the latest signals and will be with you shortly.`
   221→      const aiMsg: AgentMsg = {
   222→        id: `a-${Date.now()}`,
   223→        role: 'assistant',
   224→        content: reply,
   225→        ts: now(),
   226→      }
   227→      setHistories((h) => ({ ...h, [agent.id]: [...(h[agent.id] ?? []), aiMsg] }))
   228→    } catch {
   229→      toast.error(`${agent.name} couldn't reply. Try again.`)
   230→    } finally {
   231→      setSending(false)
   232→    }
   233→  }

   271→                  <TooltipTrigger asChild>

   459→function parseMinutesAgo(ts: string): number {
   460→  if (ts === 'now') return 0
   461→  if (ts === '—' || !ts) return Number.POSITIVE_INFINITY
   462→  const hMatch = ts.match(/(\d+)\s*h\s*(?:(\d+)\s*m)?/)
   463→  if (hMatch) {
   464→    const h = parseInt(hMatch[1]) || 0
   465→    const m = parseInt(hMatch[2] || '0') || 0
   466→    return h * 60 + m
   467→  }
   468→  const mMatch = ts.match(/(\d+)\s*min/)
   469→  if (mMatch) return parseInt(mMatch[1])
   470→  return 9999
   471→}

   649→interface ActivityItem {
   650→  id: string
   651→  agent: AIAgent | undefined
   652→  agentName: string
   653→  role: string
   654→  action: string
   655→  ts: string
   656→  kind: 'action' | 'cascade'
   657→  status?: 'done' | 'active' | 'pending'
   658→}
   659→
   660→function buildActivity(): ActivityItem[] {
   661→  const items: ActivityItem[] = []
   662→  // from agents' lastAction
   663→  AI_AGENTS.forEach((a, i) => {
   664→    items.push({
   665→      id: `act-${a.id}`,
   666→      agent: a,
   667→      agentName: a.name,
   668→      role: a.role,
   669→      action: a.lastAction,
   670→      ts: `${i * 5 + 4} min ago`,
   671→      kind: 'action',
   672→    })
   673→  })
   674→  // from cascade steps (exclude pending '—')
   675→  CASCADES.forEach((c) => {
   676→    c.steps.forEach((s, i) => {
   677→      if (s.timestamp === '—' || !s.timestamp) return
   678→      items.push({
   679→        id: `cas-${c.id}-${i}`,
   680→        agent: AGENT_BY_ID[s.agentId],
   681→        agentName: s.agent,
   682→        role: s.role,
   683→        action: s.action,
   684→        ts: s.timestamp,
   685→        kind: 'cascade',
   686→        status: s.status,
   687→      })
   688→    })
   689→  })
   690→  return items.sort((a, b) => parseMinutesAgo(a.ts) - parseMinutesAgo(b.ts))
   691→}
