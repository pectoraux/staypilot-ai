    45→/** Convert a 2-letter ISO country code to a flag emoji via regional indicator symbols. */
    46→function flagEmoji(code?: string): string {
    47→  if (!code || code.length !== 2) return '🏳️'
    48→  const cc = code.toUpperCase()
    49→  if (!/^[A-Z]{2}$/.test(cc)) return '🏳️'
    50→  return String.fromCodePoint(
    51→    ...[...cc].map((c) => 0x1f1e6 + (c.charCodeAt(0) - 65)),
    52→  )
    53→}

   298→                      className="absolute -right-0.5 -bottom-0.5 size-2.5 rounded-full border border-background"

   360→  const ota = isOta(guest.bookingSource)

   362→    () => RESERVATIONS.filter((r) => r.guestId === guest.id),
   363→    [guest.id],
   364→  )
   365→
   366→  // AI-suggested auto tags based on guest signals
   367→  const aiSuggestedTags = React.useMemo(() => {
   368→    const tags: string[] = []
   369→    if (ota) tags.push('OTA conversion opportunity')
   370→    if (guest.loyaltyTier === 'VIP' || guest.loyaltyTier === 'Gold') tags.push('High-value retention')
   371→    if (guest.lastStay) {
   372→      const days = Math.round((Date.now() - new Date(guest.lastStay + 'T00:00:00').getTime()) / 86400000)
   373→      if (days > 180) tags.push('Lapsed — win-back')
   374→      else if (days < 60) tags.push('Recent guest')
   375→    }
   376→    if (guest.travelReason === 'Business') tags.push('Business traveler')
   377→    if (guest.countryCode !== 'GH') tags.push('International outreach')

   379→      const [m, d] = guest.birthday.split('-')
   380→      const today = new Date()
   381→      const bm = Number(m), bd = Number(d)
   382→      const next = new Date(today.getFullYear(), bm - 1, bd)
   383→      if (next < today) next.setFullYear(today.getFullYear() + 1)
   384→      const daysToBday = Math.round((next.getTime() - today.getTime()) / 86400000)
   385→      if (daysToBday <= 45) tags.push('Birthday soon 🎂')

   616→export function GuestsModule() {
   617→  const { selectedGuestId, openGuest } = useApp()
   618→  const [search, setSearch] = React.useState('')
   619→  const [tierFilter, setTierFilter] = React.useState<string>('all')
   620→  const [segmentFilter, setSegmentFilter] = React.useState<string>('all')
   621→  const [mobileView, setMobileView] = React.useState<'list' | 'detail'>(
   622→    selectedGuestId ? 'detail' : 'list',
   623→  )
   624→
   625→  // Default to highest lifetimeSpend guest when nothing selected (desktop auto-pick)
   626→  const defaultGuestId = React.useMemo(
   627→    () => [...GUESTS].sort((a, b) => b.lifetimeSpend - a.lifetimeSpend)[0]?.id,
   628→    [],
   629→  )
   630→  const effectiveId = selectedGuestId ?? defaultGuestId
   631→
   632→  // If a guest was opened externally (e.g. from dashboard), switch mobile to detail
   633→  React.useEffect(() => {
   634→    if (selectedGuestId) setMobileView('detail')
   635→  }, [selectedGuestId])
   636→
   637→  const filtered = React.useMemo(() => {
   638→    const q = search.trim().toLowerCase()
   639→    return GUESTS.filter((g) => {
   640→      if (q && !g.name.toLowerCase().includes(q)) return false
   641→      if (tierFilter !== 'all' && g.loyaltyTier !== tierFilter) return false
   642→      if (segmentFilter !== 'all' && !g.segments.includes(segmentFilter as Guest['segments'][number])) return false
   643→      return true
   644→    }).sort((a, b) => b.lifetimeSpend - a.lifetimeSpend)
   645→  }, [search, tierFilter, segmentFilter])
   646→
   647→  const selectedGuest = React.useMemo(
   648→    () => GUESTS.find((g) => g.id === effectiveId) ?? null,
   649→    [effectiveId],
   650→  )
   651→
   652→  const handleSelect = (id: string) => {
   653→    openGuest(id)
   654→    setMobileView('detail')
   655→  }
   656→
   657→  const handleBack = () => setMobileView('list')
   658→
   659→  return (
   660→    <div className="space-y-4">
   661→      <SonnerToaster position="top-right" richColors closeButton />
   662→      <SectionHeader
   663→        title="Guests CRM"
   664→        description="AI-enriched guest profiles with full interaction timeline"
   665→        action={
   666→          <div className="hidden items-center gap-2 rounded-full border border-border bg-card/60 px-3 py-1.5 text-xs text-muted-foreground sm:inline-flex">
   667→            <Users className="size-3.5 text-orange-500" />
   668→            <span className="font-medium text-foreground">{GUESTS.length}</span> total guests
   669→            <span className="text-border">·</span>
   670→            <span className="font-medium text-foreground">{GUESTS.filter((g) => g.loyaltyTier === 'VIP').length}</span> VIP
   671→          </div>
   672→        }
   673→      />
   674→
   675→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-[360px_1fr]">
   676→        {/* LEFT: guest list */}
   677→        <Card
   678→          className={cn(
   679→            'flex flex-col p-3 lg:max-h-[calc(100vh-12rem)]',
   680→            mobileView === 'detail' ? 'hidden lg:flex' : 'flex',
   681→          )}
   682→        >
   683→          {/* filters */}
   684→          <div className="space-y-2 pb-3">
   685→            <div className="relative">
   686→              <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   687→              <Input
   688→                value={search}
   689→                onChange={(e) => setSearch(e.target.value)}
   690→                placeholder="Search guests by name…"
   691→                className="pl-9"
   692→              />
   693→            </div>
   694→            <div className="grid grid-cols-2 gap-2">
   695→              <Select value={tierFilter} onValueChange={setTierFilter}>
   696→                <SelectTrigger size="sm" className="w-full">
   697→                  <SelectValue placeholder="Tier" />
   698→                </SelectTrigger>
   699→                <SelectContent>
   700→                  <SelectItem value="all">All tiers</SelectItem>
   701→                  {TIERS.map((t) => (
   702→                    <SelectItem key={t} value={t}>{t}</SelectItem>
   703→                  ))}
   704→                </SelectContent>
   705→              </Select>
   706→              <Select value={segmentFilter} onValueChange={setSegmentFilter}>
   707→                <SelectTrigger size="sm" className="w-full">
   708→                  <SelectValue placeholder="Segment" />
   709→                </SelectTrigger>
   710→                <SelectContent>
   711→                  <SelectItem value="all">All segments</SelectItem>
   712→                  {SEGMENTS.map((s) => (
   713→                    <SelectItem key={s} value={s}>{s}</SelectItem>
   714→                  ))}
   715→                </SelectContent>
   716→              </Select>
   717→            </div>
   718→            <p className="px-1 text-[11px] text-muted-foreground">
   719→              Showing <span className="font-semibold text-foreground">{filtered.length}</span> of {GUESTS.length}
   720→            </p>
   721→          </div>
   722→          <Separator className="mb-2" />
   723→          {/* list */}
   724→          <div className="min-h-0 flex-1 overflow-y-auto scroll-area-fancy pr-1">
   725→            <div className="space-y-1.5">
   726→              {filtered.map((g) => (
   727→                <GuestListItem
   728→                  key={g.id}
   729→                  guest={g}
   730→                  selected={g.id === effectiveId}
   731→                  onSelect={() => handleSelect(g.id)}
   732→                />
   733→              ))}
   734→              {filtered.length === 0 && (
   735→                <div className="flex h-32 flex-col items-center justify-center gap-1 text-center">
   736→                  <Search className="size-5 text-muted-foreground/60" />

   738→                  <Button

   740→                    variant="ghost"
   741→                    onClick={() => { setSearch(''); setTierFilter('all'); setSegmentFilter('all') }}

   745→                </div>
   746→              )}
   747→            </div>
   748→          </div>
   749→        </Card>
   750→
   751→        {/* RIGHT: profile panel */}
   752→        <div className={cn(mobileView === 'list' ? 'hidden lg:block' : 'block')}>

   767→                <p className="text-sm">Select a guest to view their profile</p>
   768→              </Card>
   769→            )}
   770→          </AnimatePresence>
   771→        </div>
   772→      </div>
   773→    </div>
   774→  )
   775→}
