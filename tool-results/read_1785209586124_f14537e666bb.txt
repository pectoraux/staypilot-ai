   133→// ---------- agent chat dialog ----------
   134→function AgentChatDialog({
   135→  agent,
   136→  open,
   137→  onOpenChange,
   138→}: {
   139→  agent: AIAgent | null
   140→  open: boolean
   141→  onOpenChange: (v: boolean) => void
   142→}) {
   143→  const [histories, setHistories] = React.useState<Record<string, AgentMsg[]>>({})
   144→  const [input, setInput] = React.useState('')
   145→  const [sending, setSending] = React.useState(false)
   146→  const scrollRef = React.useRef<HTMLDivElement>(null)
   147→
   148→  const messages = agent ? histories[agent.id] ?? [] : []
   149→
   150→  React.useEffect(() => {
   151→    const el = scrollRef.current
   152→    if (el) el.scrollTop = el.scrollHeight
   153→  }, [messages.length, sending, agent?.id])
   154→
   155→  async function send(text: string) {
   156→    const trimmed = text.trim()
   157→    if (!trimmed || sending || !agent) return
   158→    const userMsg: AgentMsg = {
   159→      id: `u-${Date.now()}`,
   160→      role: 'user',
   161→      content: trimmed,
   162→      ts: now(),
   163→    }
   164→    const prior = histories[agent.id] ?? []
   165→    const next = [...prior, userMsg]
   166→    setHistories((h) => ({ ...h, [agent.id]: next }))
   167→    setInput('')
   168→    setSending(true)
   169→    try {
   170→      const history = next.map((m) => ({ role: m.role, content: m.content }))
   171→      const res = await fetch('/api/ai', {
   172→        method: 'POST',
   173→        headers: { 'Content-Type': 'application/json' },
   174→        body: JSON.stringify({
   175→          mode: 'agent-chat',
   176→          agentRole: `${agent.role} at ${PROPERTY.name}`,
   177→          message: trimmed,
   178→          history,
   179→        }),
   180→      })
   181→      const data = await res.json()
   182→      const reply =
   183→        data?.reply ??
   184→        'I am analyzing the latest data — please give me a moment and try again.'
   185→      const aiMsg: AgentMsg = {
   186→        id: `a-${Date.now()}`,
   187→        role: 'assistant',
   188→        content: reply,
   189→        ts: now(),
   190→      }
   191→      setHistories((h) => ({ ...h, [agent.id]: [...(h[agent.id] ?? []), aiMsg] }))
   192→    } catch {
   193→      toast.error(`${agent.name} couldn't reply. Try again.`)
   194→    } finally {
   195→      setSending(false)
   196→    }
   197→  }
   198→
   199→  function reset() {
   200→    if (!agent) return
   201→    setHistories((h) => {
   202→      const copy = { ...h }
   203→      delete copy[agent.id]
   204→      return copy
   205→    })
   206→    toast.success(`Cleared conversation with ${agent.name}`)
   207→  }
   208→
   209→  return (
   210→    <Dialog open={open} onOpenChange={onOpenChange}>
   211→      <DialogContent className="flex max-h-[88vh] flex-col gap-0 p-0 sm:max-w-2xl">
   212→        {agent && (
   213→          <>
   214→            <DialogHeader className="flex-row items-center gap-3 border-b border-border/60 px-4 py-3 text-left">
   215→              <div
   216→                className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl text-2xl"
   217→                style={{
   218→                  backgroundImage: `linear-gradient(135deg, ${agent.color}33, ${agent.color}11)`,
   219→                  border: `1px solid ${agent.color}40`,
   220→                }}
   221→              >
   222→                <span>{agent.avatar}</span>
   223→              </div>
   224→              <div className="min-w-0 flex-1">
   225→                <DialogTitle className="flex items-center gap-2 text-base">
   226→                  {agent.name}
   227→                  <StatusPill status={agent.status} />
   228→                </DialogTitle>
   229→                <DialogDescription className="text-xs">
   230→                  {agent.role} · {PROPERTY.name}
   231→                </DialogDescription>
   232→              </div>
   233→              <TooltipProvider>
   234→                <Tooltip>
   235→                  <TooltipTrigger asChild>
   236→                    <Button
   237→                      size="icon"
   238→                      variant="ghost"
   239→                      className="h-8 w-8"
   240→                      onClick={reset}
   241→                    >
   242→                      <RefreshCw className="h-3.5 w-3.5" />
   243→                    </Button>
   244→                  </TooltipTrigger>
   245→                  <TooltipContent>Reset conversation</TooltipContent>
   246→                </Tooltip>
   247→              </TooltipProvider>
   248→            </DialogHeader>
   249→
   250→            <p className="border-b border-border/40 bg-muted/30 px-4 py-2 text-xs italic text-muted-foreground">
   251→              {agent.description}
   252→            </p>
   253→
   254→            {/* messages */}
   255→            <div
   256→              ref={scrollRef}
   257→              className="scroll-area-fancy flex-1 space-y-3 overflow-y-auto px-4 py-4"
   258→              style={{ minHeight: '260px', maxHeight: '46vh' }}
   259→            >
   260→              {messages.length === 0 && (
   261→                <div className="flex flex-col items-center justify-center gap-2 py-10 text-center">
   262→                  <div
   263→                    className="flex h-14 w-14 items-center justify-center rounded-2xl text-3xl"
   264→                    style={{
   265→                      backgroundImage: `linear-gradient(135deg, ${agent.color}33, ${agent.color}11)`,
   266→                      border: `1px solid ${agent.color}40`,
   267→                    }}
   268→                  >
   269→                    {agent.avatar}
   270→                  </div>
   271→                  <p className="text-sm font-semibold">
   272→                    Chat with {agent.name}
   273→                  </p>
   274→                  <p className="max-w-xs text-xs text-muted-foreground">
   275→                    Ask about {agent.role.toLowerCase()} work — pricing,
   276→                    campaigns, retention, or any data point.
   277→                  </p>
   278→                  <div className="mt-2 flex flex-wrap justify-center gap-2">
   279→                    {SUGGESTED_QUESTIONS[agent.role]?.map((q) => (
   280→                      <button
   281→                        key={q}
   282→                        onClick={() => send(q)}
   283→                        disabled={sending}

   292→              <AnimatePresence initial={false}>

   299→                      'flex w-full',
   300→                      m.role === 'user' ? 'justify-end' : 'justify-start',
   301→                    )}
   302→                  >
   303→                    <div
   304→                      className={cn(
   305→                        'max-w-[82%] rounded-2xl px-3 py-2 text-sm',
   306→                        m.role === 'user'
   307→                          ? 'rounded-tr-sm bg-primary text-primary-foreground'
   308→                          : 'rounded-tl-sm bg-muted text-foreground',
   309→                      )}
   310→                    >
   311→                      {m.role === 'assistant' && (
   312→                        <div className="mb-1 flex items-center gap-1.5">
   313→                          <span
   314→                            className="inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-[10px] font-semibold"
   315→                            style={{
   316→                              backgroundColor: `${agent.color}22`,
   317→                              color: agent.color,
   318→                            }}
   319→                          >
   320→                            <Sparkles className="h-2.5 w-2.5" /> {agent.name}
   321→                          </span>
   322→                        </div>
   323→                      )}
   324→                      <p className="whitespace-pre-wrap break-words leading-relaxed">
   325→                        {m.content}
   326→                      </p>
   327→                      <p
   328→                        className={cn(
   329→                          'mt-1 text-right text-[10px]',
   330→                          m.role === 'user' ? 'text-primary-foreground/70' : 'text-muted-foreground',
   331→                        )}
   332→                      >
   333→                        {m.ts}
   334→                      </p>
   335→                    </div>
   336→                  </motion.div>
   337→                ))}
   338→              </AnimatePresence>
   339→              {sending && (
   340→                <motion.div
   341→                  initial={{ opacity: 0, y: 6 }}
   342→                  animate={{ opacity: 1, y: 0 }}
   343→                  className="flex justify-start"
   344→                >
   345→                  <div className="flex items-center gap-2 rounded-2xl rounded-tl-sm bg-muted px-3 py-2.5">
   346→                    <span
   347→                      className="inline-flex items-center gap-1 rounded-full px-1.5 py-0.5 text-[10px] font-semibold"
   348→                      style={{
   349→                        backgroundColor: `${agent.color}22`,
   350→                        color: agent.color,
   351→                      }}
   352→                    >
   353→                      <Sparkles className="h-2.5 w-2.5" /> {agent.name}
   354→                    </span>

   357→                        <motion.span
   358→                          key={i}

   385→              <Button

   455→// ---------- activity feed ----------
   456→interface ActivityItem {
   457→  id: string
   458→  agent: AIAgent
   459→  action: string
   460→  ts: string
   461→  kind: 'action' | 'recommendation'
   462→  detail?: string
   463→}
   464→
   465→function buildActivity(): ActivityItem[] {
   466→  const items: ActivityItem[] = []
   467→  // from agents' lastAction
   468→  AI_AGENTS.forEach((a, i) => {
   469→    items.push({
   470→      id: `act-${a.id}`,
   471→      agent: a,
   472→      action: a.lastAction,
   473→      ts: `${i * 7 + 3} min ago`,
   474→      kind: 'action' as const,
   475→    })
   476→  })
   477→  // from recommendations
   478→  AI_RECOMMENDATIONS.slice(0, 3).forEach((r, i) => {
   479→    const agent = AI_AGENTS.find((a) => a.id === r.agentId) ?? AI_AGENTS[0]
   480→    items.push({
   481→      id: `rec-${r.id}`,
   482→      agent,
   483→      action: `Flagged: ${r.title}`,
   484→      ts: `${i * 12 + 18} min ago`,
   485→      kind: 'recommendation' as const,
   486→      detail: r.action,
   487→    })
   488→  })
   489→  // sort by a fake "minutes ago" ascending extracted from ts
   490→  return items.sort((a, b) => {
   491→    const am = parseInt(a.ts) || 0
   492→    const bm = parseInt(b.ts) || 0
   493→    return am - bm
   494→  })
   495→}

   579→      })
   580→      const data = await res.json()
   581→      setBrief(data?.reply ?? 'No brief available.')
   582→    } catch {
   583→      toast.error('Failed to generate brief. Try again.')
   584→    } finally {
   585→      setLoading(false)
   586→    }
   587→  }
   588→
   589→  return (
   590→    <Card className="relative overflow-hidden p-5">
   591→      <div className="absolute -right-10 -top-10 h-32 w-32 rounded-full bg-gradient-to-br from-amber-500/20 to-orange-500/10 blur-2xl" />
   592→      <div className="relative flex items-start justify-between gap-3">
   593→        <div className="flex items-start gap-3">

   596→          </div>
   597→          <div>
   598→            <h3 className="text-sm font-semibold">Morning Brief</h3>

   600→              Nana, your General Manager AI, synthesizes the day ahead.
   601→            </p>
   602→          </div>
   603→        </div>
   604→        <Button
   605→          size="sm"
   606→          onClick={generate}
   607→          disabled={loading}
   608→          className="shrink-0 gap-1.5 bg-gradient-to-r from-orange-500 to-amber-500 text-white hover:from-orange-600 hover:to-amber-600"
   609→        >
   610→          {loading ? (
   611→            <RefreshCw className="h-3.5 w-3.5 animate-spin" />
   612→          ) : (
   613→            <Sparkles className="h-3.5 w-3.5" />
   614→          )}
   615→          {loading ? 'Generating…' : brief ? 'Regenerate' : 'Generate today\'s brief'}
   616→        </Button>
   617→      </div>
   618→
   619→      <Separator className="my-4" />
   620→
   621→      {loading && (
   622→        <div className="space-y-2">
   623→          {[...Array(6)].map((_, i) => (
   624→            <div
   625→              key={i}
   626→              className="shimmer h-3 rounded-full"
   627→              style={{ width: `${[100, 92, 96, 88, 70, 84][i]}%` }}
   628→            />
   629→          ))}
   630→        </div>
   631→      )}
   632→
   633→      {!loading && !brief && (
   634→        <div className="flex flex-col items-center justify-center gap-2 py-8 text-center">
   635→          <div className="flex h-12 w-12 items-center justify-center rounded-full bg-muted">

   650→        >
   651→          <ReactMarkdown
   652→            components={{

   659→              h3: ({ children }) => (

   662→                </p>

   665→              ol: ({ children }) => <ol className="ml-4 list-decimal space-y-0.5">{children}</ol>,
   666→            }}
   667→          >
   668→            {brief}
   669→          </ReactMarkdown>
