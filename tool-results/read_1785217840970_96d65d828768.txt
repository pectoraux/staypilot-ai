   233→function InsightsFeed() {
   234→  const sorted = [...REPUTATION_INSIGHTS].sort((a, b) => {
   235→    // negatives first (by mentions desc), then positives
   236→    const sv = (s: ReputationInsight['severity']) => (s === 'negative' ? 0 : s === 'positive' ? 2 : 1)
   237→    if (sv(a.severity) !== sv(b.severity)) return sv(a.severity) - sv(b.severity)
   238→    return b.mentions - a.mentions
   239→  })
   240→  return (
   241→    <Card className="flex h-full flex-col gap-0 p-0">
   242→      <div className="flex items-center justify-between border-b px-5 py-3.5">
   243→        <div>
   244→          <h3 className="text-sm font-semibold">Insights feed</h3>
   245→          <p className="text-[11px] text-muted-foreground">AI-extracted from {REPUTATION_INSIGHTS.reduce((s, i) => s + i.mentions, 0)} review mentions</p>
   246→        </div>
   247→        <Badge variant="outline" className="gap-1 border-orange-500/40 text-orange-600 dark:text-orange-400">
   248→          <Sparkles className="h-3 w-3" /> {sorted.length} live
   249→        </Badge>
   250→      </div>
   251→      <ScrollArea className="max-h-[640px] flex-1 px-4 py-4">
   252→        <div className="space-y-3">
   253→          {sorted.map((it, i) => (
   254→            <InsightCard key={it.id} insight={it} index={i} />
   255→          ))}
   256→        </div>
   257→      </ScrollArea>
   258→    </Card>
   259→  )
   260→}

   409→function ImprovementProjects() {

   413→    setProjects(prev => prev.map(p => {
   414→      if (p.id !== id) return p
   415→      const next: ProjectStatus = p.status === 'Proposed' ? 'In Progress' : p.status === 'In Progress' ? 'Done' : 'Done'
   416→      return { ...p, status: next, owner: next === 'In Progress' && p.owner === 'Unassigned' ? 'Auto-assigned' : p.owner, eta: next === 'Done' ? 'Shipped' : p.eta }
   417→    }))
   418→  }
   419→
   420→  const columns: { status: ProjectStatus; title: string; desc: string }[] = [
   421→    { status: 'Proposed', title: 'Proposed', desc: 'Derived from negative insights' },
   422→    { status: 'In Progress', title: 'In Progress', desc: 'Actively being fixed' },
   423→    { status: 'Done', title: 'Done', desc: 'Shipped to production' },
   424→  ]
   425→
   426→  return (
   427→    <Card className="p-5">
   428→      <div className="flex flex-wrap items-center justify-between gap-3">
   429→        <div>
   430→          <h3 className="text-sm font-semibold">Improvement projects</h3>
