   147→// ----------------- tiny inline sparkline (SVG) -----------------
   148→function Sparkline({ color, trend }: { color: string; trend: number }) {
   149→  // deterministic 7-point shape derived from trend sign
   150→  const pts = Array.from({ length: 7 }, (_, i) => {
   151→    const seed = (i + 1) * 9301 + Math.abs(trend) * 49297
   152→    const r = ((seed % 233280) / 233280) * 2 - 1
   153→    return 50 + r * 18 + (trend >= 0 ? i * 2.5 : -i * 2.5)
   154→  })
   155→  const w = 56
   156→  const h = 18
   157→  const d = pts.map((p, i) => {
   158→    const x = (i / (pts.length - 1)) * w
   159→    const y = h - (Math.max(20, Math.min(80, p)) / 100) * h
   160→    return `${i === 0 ? 'M' : 'L'}${x.toFixed(1)},${y.toFixed(1)}`
   161→  }).join(' ')
   162→  const gid = `spark-${color.replace('#', '')}`
   163→  return (
   164→    <svg width={w} height={h} className="opacity-80">
   165→      <defs>
   166→        <linearGradient id={gid} x1="0" y1="0" x2="0" y2="1">
   167→          <stop offset="0%" stopColor={color} stopOpacity={0.35} />
   168→          <stop offset="100%" stopColor={color} stopOpacity={0} />
   169→        </linearGradient>
   170→      </defs>
   171→      <path d={`${d} L${w},${h} L0,${h} Z`} fill={`url(#${gid})`} />
   172→      <path d={d} fill="none" stroke={color} strokeWidth={1.5} strokeLinecap="round" strokeLinejoin="round" />
   173→    </svg>
   174→  )
   175→}

   177→// ----------------- animated big number -----------------
   178→function AnimatedNumber({ value, format }: { value: number; format: (n: number) => string }) {
   179→  const [display, setDisplay] = React.useState(0)
   180→  const ref = React.useRef<number | null>(null)
   181→  React.useEffect(() => {
   182→    let raf = 0
   183→    const start = performance.now()
   184→    const from = 0
   185→    const to = value
   186→    const dur = 900
   187→    const tick = (now: number) => {
   188→      const t = Math.min(1, (now - start) / dur)
   189→      const eased = 1 - Math.pow(1 - t, 3)
   190→      setDisplay(Math.round(from + (to - from) * eased))
   191→      if (t < 1) raf = requestAnimationFrame(tick)
   192→    }
   193→    raf = requestAnimationFrame(tick)
   194→    ref.current = raf
   195→    return () => cancelAnimationFrame(raf)
   196→  }, [value])
   197→  return <>{format(display)}</>
   198→}
