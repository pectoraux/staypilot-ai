   125→// ---------- RESERVATIONS ----------
   126→const STATUS_WEIGHTS: Array<[Reservation['status'], number]> = [
   127→  ['Checked-out', 45], ['Confirmed', 30], ['Checked-in', 12], ['Cancelled', 8], ['Pending', 3], ['No-show', 2],
   128→]
   129→function weightedStatus(): Reservation['status'] {
   130→  const total = STATUS_WEIGHTS.reduce((s, [, w]) => s + w, 0)
   131→  let r = rand() * total
   132→  for (const [st, w] of STATUS_WEIGHTS) { if ((r -= w) <= 0) return st }
   133→  return 'Confirmed'
   134→}
   135→function nightsRate(room: Room, nights: number) { return room.baseRate * nights }
   136→
   137→function genReservation(i: number): Reservation {
   138→  const room = ROOMS[i % ROOMS.length]
   139→  const guest = GUESTS[i % GUESTS.length]
   140→  // spread reservations across -60..+45 days
   141→  const startOffset = randInt(-60, 45)
   142→  const nights = randInt(1, 6)
   143→  const checkIn = daysFromNow(startOffset)
   144→  const checkOut = daysFromNow(startOffset + nights)
   145→  const source = guest.bookingSource
   146→  const status = weightedStatus()
   147→  const commissionRate = source === 'Airbnb' ? 0.15 : source === 'Booking.com' ? 0.15 : source === 'Expedia' ? 0.18 : source === 'Agoda' ? 0.17 : source === 'Vrbo' ? 0.1 : 0
   148→  const gross = Math.round(nightsRate(room, nights) * (0.9 + rand() * 0.3))
   149→  const commission = Math.round(gross * commissionRate)
   150→  const acqCost = commissionRate > 0 ? Math.round(gross * (0.02 + rand() * 0.04)) : 0
   151→  return {
   152→    id: `res-${i + 1}`,
   153→    guestId: guest.id,
   154→    guestName: guest.name,
   155→    roomIds: [room.id],
   156→    checkIn, checkOut,
   157→    status,
   158→    source,
   159→    adults: randInt(1, 2),
   160→    children: rand() > 0.7 ? randInt(1, 2) : 0,
   161→    grossRevenue: gross,
   162→    commission,
   163→    netRevenue: gross - commission,
   164→    acquisitionCost: acqCost,
   165→    campaign: rand() > 0.7 ? pick(['Ramadan Promo', 'Weekend Flash Sale', 'Corporate Q4', 'Loyalty Reboot', 'Christmas Repeat']) : undefined,
   166→    coupon: rand() > 0.85 ? pick(['STAY10', 'DIRECT15', 'WELCOME', 'VIPBACK']) : undefined,
   167→    notes: rand() > 0.7 ? pick(['Needs airport pickup', 'Celebrating anniversary', 'Business meeting nearby', 'First time in Accra', '']) : '',
   168→    createdAt: daysAgo(randInt(1, 60)),
   169→  }
   170→}

   173→// ---------- TIMELINE (for a sample guest) ----------
   174→export function timelineForGuest(guestId: string): TimelineEntry[] {
   175→  const guest = GUESTS.find(g => g.id === guestId)
   176→  if (!guest) return []
   177→  const res = RESERVATIONS.filter(r => r.guestId === guestId).sort((a, b) => a.checkIn.localeCompare(b.checkIn))
   178→  const entries: TimelineEntry[] = []
   179→  res.forEach((r, idx) => {
   180→    entries.push({ id: `${guestId}-res-${idx}`, guestId, type: 'Reservation', title: `Booked ${ROOMS.find(rm => rm.id === r.roomIds[0])?.name ?? 'a room'}`, description: `${r.nights ?? ''} ${r.source} · ₵${r.grossRevenue.toLocaleString()}`, date: r.createdAt, channel: r.source, value: r.grossRevenue })
   181→    if (r.status === 'Checked-out' || r.status === 'Checked-in') {
   182→      entries.push({ id: `${guestId}-ci-${idx}`, guestId, type: 'Check-in', title: 'Checked in', description: `Room ${ROOMS.find(rm => rm.id === r.roomIds[0])?.number}`, date: r.checkIn })
   183→    }
   184→    if (r.status === 'Checked-out') {
   185→      entries.push({ id: `${guestId}-co-${idx}`, guestId, type: 'Check-out', title: 'Checked out', description: 'Stay completed', date: r.checkOut })
   186→    }
   187→    if (r.commission > 0) {
   188→      entries.push({ id: `${guestId}-pay-${idx}`, guestId, type: 'Payment', title: 'Payment received', description: `₵${r.grossRevenue.toLocaleString()} via ${r.source}`, date: r.createdAt, value: r.grossRevenue })
   189→    }
   190→  })
   191→  entries.push({ id: `${guestId}-wa`, guestId, type: 'WhatsApp', title: 'WhatsApp inquiry', description: 'Asked about airport pickup and early check-in', date: daysAgo(randInt(1, 30)), channel: 'WhatsApp', sentiment: 'neutral' })
   192→  entries.push({ id: `${guestId}-email`, guestId, type: 'Email', title: 'Welcome email sent', description: 'Sent booking confirmation and local guide', date: daysAgo(randInt(1, 20)), channel: 'Email', sentiment: 'positive' })
   193→  if (guest.avgRatingGiven && guest.avgRatingGiven >= 4) {
   194→    entries.push({ id: `${guestId}-rev`, guestId, type: 'Review', title: `Left a ${guest.avgRatingGiven}★ review`, description: pick(['"Lovely staff, great location!"', '"Felt like home, will return."', '"Clean and comfortable, highly recommend."']), date: daysAgo(randInt(1, 60)), sentiment: 'positive' })
   195→  }
   196→  entries.push({ id: `${guestId}-camp`, guestId, type: 'Campaign', title: 'Added to "Loyalty Reboot" campaign', description: 'Targeted as lapsed high-value guest', date: daysAgo(randInt(1, 15)) })
   197→  entries.sort((a, b) => b.date.localeCompare(a.date))
   198→  return entries
   199→}
